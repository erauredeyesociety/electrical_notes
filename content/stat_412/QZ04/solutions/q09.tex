\section*{Problem}
The joint density is
\[
f(x,y,z)=\frac{3xy^2z}{25},
\qquad0<x<2,\quad0<y<1,\quad0<z<5,
\]
and zero elsewhere. Find (a) the joint marginal of $Y,Z$, (b) the marginal of
$Y$, (c) $P(1/4<X<5/4,Y>1/3,3<Z<4)$, and
(d) $P(0<X<3/4\mid Y=1/2,Z=2)$.

For part (a), the choices are $g(x,z)$, $g(x,y)$, or $g(y,z)$ on their
corresponding supports. For part (b), the choices are $h(x)$, $h(z)$, or
$h(y)$ on their corresponding supports.

\section*{Part (a)}
\[
\begin{aligned}
g(y,z)
&=\int_0^2\frac{3xy^2z}{25}\,dx\\
&=\frac{3y^2z}{25}\left[\frac{x^2}{2}\right]_0^2\\
&=\frac{6y^2z}{25}.
\end{aligned}
\]
\finalanswer{(a) Select Choice C:
$g(y,z)=\dfrac{6y^2z}{25}$ for $0<y<1$, $0<z<5$.}

\section*{Part (b)}
\[
\begin{aligned}
h(y)
&=\int_0^5g(y,z)\,dz\\
&=\frac{6y^2}{25}\left[\frac{z^2}{2}\right]_0^5\\
&=\frac{6y^2}{25}\left(\frac{25}{2}\right)
=3y^2.
\end{aligned}
\]
\finalanswer{(b) Select Choice C: $h(y)=3y^2$ for $0<y<1$.}

\section*{Part (c)}
\[
\begin{aligned}
P&=\int_{1/4}^{5/4}\int_{1/3}^{1}\int_3^4
\frac{3xy^2z}{25}\,dz\,dy\,dx\\
&=\frac3{25}
\left(\int_{1/4}^{5/4}x\,dx\right)
\left(\int_{1/3}^{1}y^2\,dy\right)
\left(\int_3^4z\,dz\right)\\
&=\frac3{25}
\left[\frac{x^2}{2}\right]_{1/4}^{5/4}
\left[\frac{y^3}{3}\right]_{1/3}^{1}
\left[\frac{z^2}{2}\right]_3^4\\
&=\frac3{25}\left(\frac34\right)
\left(\frac{26}{81}\right)\left(\frac72\right)\\
&=\frac{91}{900}.
\end{aligned}
\]
\finalanswer{(c) $\dfrac{91}{900}$.}

\section*{Part (d)}
For fixed $Y=1/2,Z=2$, the conditional density of $X$ is proportional to $x$:
\[
f_{X\mid Y,Z}(x\mid1/2,2)
=\frac{x}{\int_0^2u\,du}
=\frac{x}{[u^2/2]_0^2}
=\frac{x}{2},\qquad0<x<2.
\]
Therefore
\[
P(0<X<3/4\mid Y=1/2,Z=2)
=\int_0^{3/4}\frac{x}{2}\,dx
=\left[\frac{x^2}{4}\right]_0^{3/4}
=\frac9{64}.
\]
\finalanswer{(d) $\dfrac9{64}$.}

\section*{Problem}
If $\sigma_X^2=3$, $\sigma_Y^2=9$, and $\sigma_{XY}=4$, where $X$ and $Y$
are not independent, find the variance of $Z=-4X+6Y-8$.

\section*{Setup}
For dependent variables,
\[
\operatorname{Var}(aX+bY+c)
=a^2\operatorname{Var}(X)+b^2\operatorname{Var}(Y)
+2ab\operatorname{Cov}(X,Y).
\]
The constant has no effect on variance.

\section*{Work}
\[
\begin{aligned}
\sigma_Z^2
&=(-4)^2(3)+6^2(9)+2(-4)(6)(4)\\
&=48+324-192\\
&=180.
\end{aligned}
\]

\finalanswer{$\sigma_Z^2=180$.}

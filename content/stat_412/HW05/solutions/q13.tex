\section*{Problem}
Bank teller service time has $\mu=3.7$ minutes and $\sigma=1.8$ minutes. For
$n=81$, find probabilities for $\bar X$: (a) at most $3.4$, (b) more than
$3.9$, and (c) at least $3.7$ but less than $4.1$.

\section*{Work}
\[
\sigma_{\bar X}=\frac{1.8}{\sqrt{81}}=\frac{1.8}{9}=0.2.
\]
For part (a),
\[
z=\frac{3.4-3.7}{0.2}=-1.50,\qquad P(\bar X\le3.4)=0.0668.
\]
For part (b),
\[
z=\frac{3.9-3.7}{0.2}=1.00,\qquad P(\bar X>3.9)=1-0.8413=0.1587.
\]
For part (c),
\[
z_1=0,\qquad z_2=\frac{4.1-3.7}{0.2}=2.00,
\]
so
\[
P(3.7\le\bar X<4.1)=0.9772-0.5000=0.4772.
\]

\finalanswer{(a) $0.0668$; (b) $0.1587$; (c) $0.4772$.}

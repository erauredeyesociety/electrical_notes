\item \textbf{Conditional probability:}

\item \textbf{Joint density $f=\tfrac34 x$:}

\item \textbf{Covariance (machine malfunctions vs.\ calls):}

\item \textbf{Correlation (oranges vs.\ apples):}

(enter the \textbf{exact radical}; the decimal $-0.5477$ was rejected).

\item \textbf{Correlation, $f=0.5$ on $0<x\le y<2$:}

\item \textbf{Independent variables:}
(a) $E(9X-7Y)=3.8$;\quad (b) $E(XY)=E(X)E(Y)=(2.6)(2.8)=7.28$.

\item \textbf{Variance of a linear combination:}
$\sigma^2_{-8X+6Y-5}=64(6)+36(9)=708$.

\item \textbf{Two 24-sided dice:}

\item \textbf{Two 8-sided dice} (each $\operatorname{Var}=\tfrac{63}{12}=5.25$):
(a) $\operatorname{Var}(4X-Y)=16(5.25)+5.25=89.25$;\quad
(b) $\operatorname{Var}(X+3Y-5)=5.25+9(5.25)=52.5$.

\item \textbf{Independence test, $f=x^2+2y^2$:}
(a) $X,Y$ are \textbf{not} independent, since $f(x,y)$ \textbf{is not} equal to

\quad (c) $\operatorname{Var}(X)=\dfrac{59}{720}$,

\item \textbf{Chebyshev (bulb life), symmetric:}
at most $\dfrac{1}{2(4)^2}=\dfrac{1}{32}=3.125\%$ fail to last 660 hours.

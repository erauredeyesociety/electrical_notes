\section*{Problem}
$X$ and $Y$ are independent with the joint distribution
\[
\begin{array}{c|cc}
f(x,y) & x=2 & x=4\\\hline
y=1 & 0.175 & 0.075\\
y=3 & 0.42 & 0.18\\
y=5 & 0.105 & 0.045
\end{array}
\]
Find $E(9X-7Y)$.

\section*{Setup}
Expectation is linear: $E(aX+bY)=aE(X)+bE(Y)$. Get $E(X)$ and $E(Y)$ from the
marginals $g(x)=\sum_y f$, $h(y)=\sum_x f$.

\section*{Work}
\[
g(2)=0.70,\ g(4)=0.30\ \Rightarrow\ E(X)=2(0.70)+4(0.30)=2.6 .
\]
\[
h(1)=0.25,\ h(3)=0.60,\ h(5)=0.15\ \Rightarrow\ E(Y)=1(0.25)+3(0.60)+5(0.15)=2.8 .
\]
\[
E(9X-7Y)=9(2.6)-7(2.8)=23.4-19.6=3.8 .
\]

\textbf{(b)} Since $X$ and $Y$ are \emph{independent}, the expectation of the
product factors:
\[
E(XY)=E(X)\,E(Y)=(2.6)(2.8)=7.28 .
\]

\finalanswer{(a) $E(9X-7Y)=3.8$.\quad (b) $E(XY)=E(X)E(Y)=7.28$.}
